# 1. Enable SFTP on Synology DSM
Control Panel → File Services → FTP → SFTP → Enable SFTP service

# 2. Test SFTP
sftp leo_admin@192.168.1.188

# 3. Create a local mount point
mkdir -p ~/synology

# 4. Mount the Synology share with SSHFS
sshfs leo_admin@192.168.1.188:/Drive_A ~/synology

# 5. Verify the mount
ls ~/synology

# 6. Run the Rust deduper
cargo run --release -- FileDeduper -p ~/synology

# 7. Unmount when finished
fusermount3 -u ~/synology

# IMPORTANT:
# Changes made inside ~/synology affect the original files on the Synology NAS.
