cmake_minimum_required(VERSION 4.4)

project(data_structures
    VERSION 1.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(PROJECT_WARNINGS
    -Wall
    -Wextra
    -Wpedantic
)


# -------------------------
# Shared library
# -------------------------

add_library(data_structures_lib
    src/BST.cpp
)

target_include_directories(data_structures_lib
    PUBLIC
        include
)

target_compile_options(data_structures_lib
    PRIVATE
        ${PROJECT_WARNINGS}
)


# -------------------------
# Main executable
# -------------------------

add_executable(${PROJECT_NAME})

include(cmake/sources.cmake)   # generated by cppgen-cli

target_include_directories(${PROJECT_NAME}
    PRIVATE
        include
)

target_compile_options(${PROJECT_NAME}
    PRIVATE
        ${PROJECT_WARNINGS}
)


# -------------------------
# Tests
# -------------------------

enable_testing()

add_executable(tests
    tests/test_main.cpp
)

# Reuse every source cppgen added to the main target, except its main()
get_target_property(LIB_SOURCES ${PROJECT_NAME} SOURCES)
list(FILTER LIB_SOURCES EXCLUDE REGEX "src/main\\.cpp$")
target_sources(tests PRIVATE ${LIB_SOURCES})

target_include_directories(tests
    PRIVATE
        include
)

target_compile_options(tests
    PRIVATE
        ${PROJECT_WARNINGS}
)

add_test(NAME unit_tests COMMAND tests)
