cmake_minimum_required(VERSION 4.3)

project(similar_files_checker
    VERSION 1.0
    LANGUAGES C CXX
)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(PROJECT_WARNINGS
    -Wall
    -Wextra
    -Wpedantic
)


# -------------------------
# Sanitizers
# -------------------------
# These instrument this project's code only. Dependencies come prebuilt
# from vcpkg and are NOT instrumented.

option(ENABLE_ASAN "Build with AddressSanitizer" OFF)
option(ENABLE_TSAN "Build with ThreadSanitizer" OFF)

if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "ENABLE_ASAN and ENABLE_TSAN can't be used together")
endif()

if(ENABLE_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address)
endif()

if(ENABLE_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g -O1)
  add_link_options(-fsanitize=thread)
endif()


# -------------------------
# Dependencies (declared in vcpkg.json)
# -------------------------

find_package(fmt      CONFIG REQUIRED)
find_package(exiv2    CONFIG REQUIRED)
find_package(argparse CONFIG REQUIRED)
find_package(BLAKE3   CONFIG REQUIRED)
find_package(TBB      CONFIG REQUIRED)
find_package(OpenCV   REQUIRED)
find_package(FFMPEG   REQUIRED)   # vcpkg's FFmpeg find module

# Wrap FFmpeg's variables in a target so it links like everything else
add_library(ffmpeg_deps INTERFACE)
target_include_directories(ffmpeg_deps INTERFACE ${FFMPEG_INCLUDE_DIRS})
target_link_directories(ffmpeg_deps INTERFACE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(ffmpeg_deps INTERFACE ${FFMPEG_LIBRARIES})

# -------------------------
# Bounds checking (Debug only)
# -------------------------
# libstdc++ reads the first macro, libc++ the second; each ignores the other.
add_library(bounds_checks INTERFACE)
target_compile_definitions(bounds_checks INTERFACE
    _GLIBCXX_ASSERTIONS
    _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST
)

# -------------------------
# Main executable
# -------------------------

add_executable(${PROJECT_NAME})

include(cmake/sources.cmake)   # generated by cppgen-cli

target_include_directories(${PROJECT_NAME}
    PRIVATE
        include
        ${OpenCV_INCLUDE_DIRS}
)

target_compile_options(${PROJECT_NAME}
    PRIVATE
        ${PROJECT_WARNINGS}
)

target_link_libraries(${PROJECT_NAME}
    PRIVATE
        fmt::fmt
        ${OpenCV_LIBS}
        Exiv2::exiv2lib
        ffmpeg_deps
        argparse::argparse
        TBB::tbb
        BLAKE3::blake3
        bounds_checks
)


# -------------------------
# Tests
# -------------------------

enable_testing()

# Reuse the main target's sources, minus main.cpp
get_target_property(APP_SOURCES ${PROJECT_NAME} SOURCES)
list(FILTER APP_SOURCES EXCLUDE REGEX "main\\.cpp$")

add_executable(tests
    tests/test_main.cpp
    ${APP_SOURCES}
)

target_include_directories(tests
    PRIVATE
        include
        ${OpenCV_INCLUDE_DIRS}
)

target_compile_options(tests
    PRIVATE
        ${PROJECT_WARNINGS}
)

target_compile_definitions(tests
    PRIVATE
        TEST_DATA_DIR="${CMAKE_SOURCE_DIR}/tests/directory_contents"
)

target_link_libraries(tests
    PRIVATE
        fmt::fmt
        ${OpenCV_LIBS}
        Exiv2::exiv2lib
        ffmpeg_deps
        argparse::argparse
        BLAKE3::blake3
        TBB::tbb
        bounds_checks
)

add_test(NAME unit_tests COMMAND tests)
